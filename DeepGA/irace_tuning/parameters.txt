# --- Variante de Neuroevolución (Solo V10, V11 y V12) ---

# lr: Tasa de aprendizaje de Adam en escala logarítmica (r,log)

# --- Parámetros Poblacionales & Evolutivos Globales ---
pop_size               "--pop-size "               i       (8, 40)
generations            "--generations "            i       (5, 50)
t_size                 "--t-size "                 i       (2, 8)
cr                     "--cr "                     r       (0.50, 0.95)
mr                     "--mr "                     r       (0.20, 0.70)

mr_max                 "--mr-max "                 r       (0.65, 0.95)
final_epoch            "--final-epochs "           i       (5, 30)

# --- Meta-Modelo Subrogado (Random Forest + UCB) ---
pool_candidates_factor "--pool-candidates-factor " i       (2, 6)
kappa                  "--kappa "                  r       (0.05, 0.30)

# ==============================================================================
# Parámetros Condicionales por Variante
